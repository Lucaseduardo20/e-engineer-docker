```typescript
import { httpClient } from '../http-client'

export class BaseApiService {
  constructor(protected readonly baseUrl: string) {}

  protected getUrl(path: string): string {
    return `${this.baseUrl}${path}`
  }

  protected async get<T>(path: string) {
    const { data } = await httpClient.get<{ data: T }>(this.getUrl(path))
    return data
  }

  protected async post<T>(path: string, payload: any) {
    const { data } = await httpClient.post<{ data: T }>(this.getUrl(path), payload)
    return data
  }

  protected async put<T>(path: string, payload: any) {
    const { data } = await httpClient.put<{ data: T }>(this.getUrl(path), payload)
    return data
  }

  protected async delete<T>(path: string) {
    const { data } = await httpClient.delete<{ data: T }>(this.getUrl(path))
    return data
  }
}
```

#### B) `e-engineer-frontend/src/shared/http/api/auth.ts`
```typescript
import { httpClient } from '../http-client'

export interface LoginRequest {
  email: string
  password: string
}

export interface LoginResponse {
  token: string
  user: {
    id: string
    email: string
    name: string
    organizationId: string
  }
}

export class AuthService {
  async login(request: LoginRequest): Promise<LoginResponse> {
    const { data } = await httpClient.post<{ data: LoginResponse }>(
      '/auth/login',
      request
    )
    return data
  }

  async refreshToken(token: string): Promise<{ token: string }> {
    const { data } = await httpClient.post<{ data: { token: string } }>(
      '/auth/refresh',
      { token }
    )
    return data
  }
}

export const authService = new AuthService()
```

#### C) `e-engineer-frontend/src/shared/http/api/projects.ts`
```typescript
import { BaseApiService } from './base'

export interface ProjectDTO {
  id: string
  name: string
  type: string
  status: 'draft' | 'active' | 'archived'
  organizationId: string
  createdAt: string
  updatedAt: string
}

export interface CreateProjectRequest {
  name: string
  type: string
  description?: string
}

export interface ListProjectsResponse {
  data: ProjectDTO[]
  total: number
  page: number
  limit: number
}

export class ProjectsService extends BaseApiService {
  constructor() {
    super('/projects')
  }

  async list(page = 1, limit = 10): Promise<ListProjectsResponse> {
    const response = await this.get<ListProjectsResponse>(
      `?page=${page}&limit=${limit}`
    )
    return response as any
  }

  async getById(id: string): Promise<ProjectDTO> {
    return this.get<ProjectDTO>(`/${id}`)
  }

  async create(request: CreateProjectRequest): Promise<ProjectDTO> {
    return this.post<ProjectDTO>('', request)
  }

  async update(id: string, request: Partial<CreateProjectRequest>): Promise<ProjectDTO> {
    return this.put<ProjectDTO>(`/${id}`, request)
  }

  async delete(id: string): Promise<void> {
    await this.delete<void>(`/${id}`)
  }
}

export const projectsService = new ProjectsService()
```

```typescript
import { projectsService } from '@/shared/http/api'

export const useProjectsStore = defineStore('projects', () => {
  const projects = ref<ProjectDTO[]>([])
  const loading = ref(false)

  const fetchProjects = async (page = 1, limit = 10) => {
    try {
      loading.value = true
      const response = await projectsService.list(page, limit)
      projects.value = response.data
    } catch (error) {
      console.error('Erro ao buscar projetos:', error)
    } finally {
      loading.value = false
    }
  }

  const createProject = async (name: string, type: string) => {
    const project = await projectsService.create({ name, type })
    projects.value.push(project)
    return project
  }

  return { projects, loading, fetchProjects, createProject }
})
```

### ✅ Task 8: Interceptor Refresh Token [1-2 horas]

**Editar:** `e-engineer-frontend/src/shared/http/interceptors/auth.interceptor.ts`

Adicionar response interceptor:
```typescript
httpClient.interceptors.response.use(
  response => response,
  async error => {
    const originalRequest = error.config

    if (error.response?.status === 401 && !originalRequest._retry) {
      originalRequest._retry = true

      try {
        const token = localStorage.getItem('auth:token')
        if (token) {
          const { token: newToken } = await authService.refreshToken(token)
          localStorage.setItem('auth:token', newToken)
          originalRequest.headers.Authorization = `Bearer ${newToken}`
          return httpClient(originalRequest)
        }
      } catch {
        // Logout
        authStore.clear()
        router.push('/login')
      }
    }

    return Promise.reject(error)
  }
)
```

```typescript
import { test, expect } from '@playwright/test'

const BASE_URL = 'http://localhost:5173'
const API_URL = 'http://localhost:3000'

test.describe('Sprint 0 - MVP Integration', () => {
  test('E2E: Login → Dashboard → Create Project → Logout', async ({ page }) => {
    // 1. Login
    await page.goto(`${BASE_URL}/login`)
    await page.fill('[name="email"]', 'admin@engflow.local')
    await page.fill('[name="password"]', '123456')
    await page.click('button:has-text("Entrar")')

    // 2. Esperar dashboard
    await page.waitForURL(`${BASE_URL}/dashboard`)
    await expect(page.locator('h1')).toContainText('Dashboard')

    // 3. Validar projetos reais
    const projectsList = page.locator('[data-testid="projects-list"]')
    await expect(projectsList).toBeVisible()
    const projectCount = await projectsList.locator('tr').count()
    expect(projectCount).toBeGreaterThan(0)

    // 4. Criar projeto
    await page.click('[data-testid="btn-create-project"]')
    await page.fill('[name="name"]', 'Projeto E2E Test')
    await page.fill('[name="type"]', 'Estrutura')
    await page.click('button:has-text("Salvar")')

    // 5. Validar que foi criado
    await page.waitForTimeout(500)
    await expect(page.locator('text=Projeto E2E Test')).toBeVisible()

    // 6. Logout
    await page.click('[data-testid="btn-logout"]')
    await page.waitForURL(`${BASE_URL}/login`)

    // 7. Validar sessão foi limpa
    const token = await page.evaluate(() => localStorage.getItem('auth:token'))
    expect(token).toBeNull()
  })
})
```
