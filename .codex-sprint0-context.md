#### **3.1: Criar `e-engineer-frontend/src/shared/http/api/base.ts`**

```typescript
import { httpClient } from '../http-client'

export class BaseApiService {
  constructor(
    protected readonly baseUrl: string,
    private readonly organizationId?: string
  ) {}

  protected getUrl(path: string): string {
    return `${this.baseUrl}${path}`
  }

  protected async get<T>(path: string, config = {}) {
    return httpClient.get<T>(this.getUrl(path), config)
  }

  protected async post<T>(path: string, data?: any, config = {}) {
    return httpClient.post<T>(this.getUrl(path), data, config)
  }

  protected async put<T>(path: string, data?: any, config = {}) {
    return httpClient.put<T>(this.getUrl(path), data, config)
  }

  protected async delete<T>(path: string, config = {}) {
    return httpClient.delete<T>(this.getUrl(path), config)
  }
}
```

#### **3.2: Criar `e-engineer-frontend/src/shared/http/api/auth.ts`**

```typescript
import { httpClient } from '../http-client'

export interface LoginRequest {
  email: string
  password: string
}

export interface LoginResponse {
  token: string
  user: {
    id: string
    email: string
    name: string
    organizationId: string
  }
}

export class AuthService {
  private readonly baseUrl = '/auth'

  async login(request: LoginRequest): Promise<LoginResponse> {
    const response = await httpClient.post<any>(
      `${this.baseUrl}/login`,
      request
    )
    return response.data
  }

  async refreshToken(token: string): Promise<{ token: string }> {
    const response = await httpClient.post<any>(
      `${this.baseUrl}/refresh`,
      { token }
    )
    return response.data
  }
}

export const authService = new AuthService()
```

#### **3.3: Criar `e-engineer-frontend/src/shared/http/api/projects.ts`**

```typescript
import { BaseApiService } from './base'

export interface ProjectDTO {
  id: string
  name: string
  type: string
  status: 'draft' | 'active' | 'archived'
  description?: string
  timeline?: string
  organizationId: string
  createdAt: string
  updatedAt: string
}

export interface CreateProjectRequest {
  name: string
  type: string
  description?: string
  timeline?: string
}

export interface ListProjectsResponse {
  data: ProjectDTO[]
  total: number
  page: number
  limit: number
}

export class ProjectsService extends BaseApiService {
  constructor() {
    super('/projects')
  }

  async list(
    page = 1,
    limit = 10,
    filters?: { status?: string; name?: string }
  ): Promise<ListProjectsResponse> {
    const params = new URLSearchParams()
    params.append('page', page.toString())
    params.append('limit', limit.toString())

    if (filters?.status) params.append('status', filters.status)
    if (filters?.name) params.append('name', filters.name)

    const response = await this.get<any>(
      `?${params.toString()}`
    )
    return response.data
  }

  async getById(id: string): Promise<ProjectDTO> {
    const response = await this.get<any>(`/${id}`)
    return response.data
  }

  async create(request: CreateProjectRequest): Promise<ProjectDTO> {
    const response = await this.post<any>('', request)
    return response.data
  }

  async update(id: string, request: Partial<CreateProjectRequest>): Promise<ProjectDTO> {
    const response = await this.put<any>(`/${id}`, request)
    return response.data
  }

  async delete(id: string): Promise<void> {
    await this.delete<void>(`/${id}`)
  }
}

export const projectsService = new ProjectsService()
```

**Exemplo:**
```typescript
import { projectsService } from '@/shared/http/api'

export const useProjectsStore = defineStore('projects', () => {
  const projects = ref<ProjectDTO[]>([])
  const loading = ref(false)
  const error = ref<string | null>(null)
  const pagination = reactive({
    page: 1,
    limit: 10,
    total: 0
  })

  const fetchProjects = async (filters?: any) => {
    try {
      loading.value = true
      error.value = null
      const response = await projectsService.list(
        pagination.page,
        pagination.limit,
        filters
      )
      projects.value = response.data
      pagination.total = response.total
    } catch (err) {
      error.value = err.message
    } finally {
      loading.value = false
    }
  }

  const createProject = async (data: CreateProjectRequest) => {
    try {
      const newProject = await projectsService.create(data)
      projects.value.push(newProject)
      return newProject
    } catch (err) {
      throw err
    }
  }

  return {
    projects,
    loading,
    error,
    pagination,
    fetchProjects,
    createProject
  }
})
```

### Task 8: Atualizar Axios Interceptor para renovar token automaticamente (1-2 horas) 🟡

**Arquivo:** `e-engineer-frontend/src/shared/http/interceptors/auth.interceptor.ts`

**O que fazer:**
1. Capturar response com status 401
2. Tentar renovar token com `/auth/refresh`
3. Se sucesso, retry requisição original
4. Se erro, fazer logout

**Exemplo:**
```typescript
httpClient.interceptors.response.use(
  response => response,
  async error => {
    const originalRequest = error.config

    if (error.response?.status === 401 && !originalRequest._retry) {
      originalRequest._retry = true

      try {
        const token = localStorage.getItem('auth:token')
        const { token: newToken } = await authService.refreshToken(token)
        localStorage.setItem('auth:token', newToken)

        originalRequest.headers.Authorization = `Bearer ${newToken}`
        return httpClient(originalRequest)
      } catch (err) {
        // Logout
        authStore.clear()
        router.push('/login')
      }
    }

    return Promise.reject(error)
  }
)
```

```typescript
import { test, expect } from '@playwright/test'

test.describe('Auth Flow E2E', () => {
  test('Login → Dashboard → Projects → Logout', async ({ page }) => {
    // 1. Ir para login
    await page.goto('http://localhost:5173/login')
    await expect(page).toHaveTitle(/Login/)

    // 2. Fazer login
    await page.fill('input[name="email"]', 'admin@engflow.local')
    await page.fill('input[name="password"]', '123456')
    await page.click('button:has-text("Entrar")')

    // 3. Esperar redirecionar para dashboard
    await page.waitForURL('**/dashboard')
    await expect(page.locator('h1')).toContainText('Dashboard')

    // 4. Validar que projetos aparecem
    const projectsList = page.locator('[data-testid="projects-list"]')
    await expect(projectsList).toBeVisible()
    
    const projectItems = projectsList.locator('[data-testid="project-item"]')
    const count = await projectItems.count()
    expect(count).toBeGreaterThan(0)

    // 5. Fazer logout
    await page.click('[data-testid="logout-button"]')
    await page.waitForURL('**/login')
    
    // 6. Validar que token foi removido
    const token = await page.evaluate(() => localStorage.getItem('auth:token'))
    expect(token).toBeNull()
  })
})
```
